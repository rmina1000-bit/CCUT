"""
preview_clip_engine.py
======================
Fragment preview clip 생성 엔진.

목적:
  - CenterPanel이 원본 영상 + currentTime seek 방식을 버리고,
    fragment마다 독립적인 mp4 clip을 재생하도록 전환한다.
  - ensure_preview_clip() 호출 시:
      clip이 이미 있으면 → 재사용 (idempotent)
      clip이 없으면    → ffmpeg re-encode로 생성

금지:
  - -c copy 사용 금지 (faststart + 독립 keyframe 보장을 위해 re-encode 필수)
  - video_url fallback 금지 (clip 없으면 MISSING 반환)
"""

import os
import subprocess
import logging
from pathlib import Path

logger = logging.getLogger(__name__)

# STORAGE_DIR은 main.py에서 환경변수로 결정됨.
# 직접 import 시 fallback으로 ccut_backend/../storage를 사용.
_DEFAULT_STORAGE = Path(__file__).resolve().parent.parent.parent / "storage"


def _get_preview_clips_dir(storage_dir: Path) -> Path:
    d = storage_dir / "preview_clips"
    d.mkdir(parents=True, exist_ok=True)
    return d


def ensure_preview_clip(
    fragment_id: str,
    source_path: str,
    start: float,
    end: float,
    storage_dir: Path | None = None,
    app_base_url: str = "http://127.0.0.1:8000",
) -> dict:
    """
    fragment preview clip이 없으면 ffmpeg로 생성, 있으면 재사용.
    """
    if storage_dir is None:
        env_dir = os.getenv("CCUT_STORAGE_DIR")
        storage_dir = Path(env_dir) if env_dir else _DEFAULT_STORAGE

    clips_dir = _get_preview_clips_dir(storage_dir)
    output_path = clips_dir / f"{fragment_id}.mp4"
    duration = max(0.0, end - start)

    app_base_url = app_base_url.rstrip("/")
    clip_url = f"{app_base_url}/static/preview_clips/{fragment_id}.mp4"

    if output_path.exists() and output_path.stat().st_size > 1024:
        return {"fragment_id": fragment_id, "preview_clip_url": clip_url,
                "duration": duration, "status": "CACHED", "error": None}

    if not source_path or not os.path.exists(source_path):
        msg = f"[PREVIEW_CLIP_ENGINE] source_path not found: {source_path!r}"
        return {"fragment_id": fragment_id, "preview_clip_url": None,
                "duration": duration, "status": "ERROR", "error": msg}

    if duration < 0.1:
        msg = f"[PREVIEW_CLIP_ENGINE] duration too short ({duration}s)"
        return {"fragment_id": fragment_id, "preview_clip_url": None,
                "duration": duration, "status": "ERROR", "error": msg}

    cmd = [
        "ffmpeg", "-y",
        "-ss", str(start), "-to", str(end), "-i", str(source_path),
        "-vf", "scale=-2:720,fps=30",
        "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
        "-c:a", "aac", "-b:a", "128k",
        "-movflags", "+faststart",
        str(output_path),
    ]

    try:
        result = subprocess.run(cmd, stdout=subprocess.PIPE,
                                stderr=subprocess.PIPE, timeout=120)
        if result.returncode != 0:
            stderr_text = result.stderr.decode("utf-8", errors="replace")[-2000:]
            msg = f"[PREVIEW_CLIP_ENGINE] ffmpeg FAILED (rc={result.returncode}): {stderr_text}"
            if output_path.exists():
                output_path.unlink(missing_ok=True)
            return {"fragment_id": fragment_id, "preview_clip_url": None,
                    "duration": duration, "status": "ERROR", "error": msg}

        if not output_path.exists() or output_path.stat().st_size < 1024:
            msg = f"[PREVIEW_CLIP_ENGINE] output file missing or too small: {output_path}"
            return {"fragment_id": fragment_id, "preview_clip_url": None,
                    "duration": duration, "status": "ERROR", "error": msg}

        return {"fragment_id": fragment_id, "preview_clip_url": clip_url,
                "duration": duration, "status": "CREATED", "error": None}

    except subprocess.TimeoutExpired:
        if output_path.exists():
            output_path.unlink(missing_ok=True)
        return {"fragment_id": fragment_id, "preview_clip_url": None,
                "duration": duration, "status": "ERROR",
                "error": f"[PREVIEW_CLIP_ENGINE] ffmpeg TIMEOUT for {fragment_id}"}
    except Exception as e:
        return {"fragment_id": fragment_id, "preview_clip_url": None,
                "duration": duration, "status": "ERROR",
                "error": f"[PREVIEW_CLIP_ENGINE] EXCEPTION: {e}"}
